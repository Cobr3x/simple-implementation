if(NOT CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Use -DCMAKE_TOOLCHAIN_FILE=cmake/arm-none-eabi.cmake in a fresh build directory")
endif()
set(AUTH_DEPS_ROOT "${CMAKE_SOURCE_DIR}/.deps" CACHE PATH "Pinned CMSIS/HAL dependency directory")
set(HAL_DIR ${AUTH_DEPS_ROOT}/hal-${FAMILY})
set(CMSIS_DIR ${AUTH_DEPS_ROOT}/cmsis-${FAMILY})
if(NOT EXISTS "${HAL_DIR}/Src/stm32${FAMILY}xx_hal.c")
    message(FATAL_ERROR "Run python3 tools/fetch_deps.py first, or set AUTH_DEPS_ROOT")
endif()
set(HAL_MODULES hal hal_cortex hal_gpio hal_rcc hal_rcc_ex hal_pwr hal_pwr_ex
    hal_flash hal_flash_ex hal_uart hal_uart_ex hal_rng hal_dma hal_dma_ex)
if(FAMILY STREQUAL "h7")
    list(APPEND HAL_MODULES hal_spi hal_spi_ex)
endif()
set(HAL_SOURCES)
foreach(module IN LISTS HAL_MODULES)
    list(APPEND HAL_SOURCES ${HAL_DIR}/Src/stm32${FAMILY}xx_${module}.c)
endforeach()
add_library(stm32_hal STATIC ${HAL_SOURCES} ${CMSIS_DIR}/Source/Templates/system_stm32${FAMILY}xx.c)
target_include_directories(stm32_hal PUBLIC ${HAL_DIR}/Inc ${CMSIS_DIR}/Include
    ${AUTH_DEPS_ROOT}/cmsis-core/CMSIS/Core/Include ${INSTANCE_DIR} ${FAMILY_DIR})
target_compile_definitions(stm32_hal PUBLIC ${DEVICE} USE_HAL_DRIVER)
if(FAMILY STREQUAL "h7")
    target_compile_definitions(stm32_hal PUBLIC USE_PWR_LDO_SUPPLY)
endif()
add_library(auth_drivers STATIC board.c store.c)
target_include_directories(auth_drivers PUBLIC ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(auth_drivers PUBLIC auth_core stm32_hal)
target_compile_options(auth_drivers PRIVATE -Wall -Wextra -Werror)
option(AUTH_REQUIRE_PROTECTION "Require RDP1 and code/key write protection at boot" ON)
target_compile_definitions(auth_drivers PRIVATE AUTH_REQUIRE_PROTECTION=$<BOOL:${AUTH_REQUIRE_PROTECTION}>)
if(AUTH_LCD)
    target_sources(auth_drivers PRIVATE ${PROJECT_SOURCE_DIR}/drivers/components/st7735_h7.c)
endif()
